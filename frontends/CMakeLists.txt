cmake_minimum_required(VERSION 3.20)
if((CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin") AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    # Clang raises arm64 to its platform floor of 11.0; Intel retains 10.13.
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Intel macOS deployment minimum")
endif()
project(ArchetypeCabDisabler LANGUAGES C CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../shared/version.h" cab_version_line REGEX "^#define CAB_VERSION ")
string(REGEX MATCH "[0-9]+[.][0-9]+[.][0-9]+" cab_version "${cab_version_line}")
foreach(platform windows macos)
    set(notice "${CMAKE_CURRENT_SOURCE_DIR}/../licenses/${platform}-LICENSES.txt")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${notice}" "${CMAKE_CURRENT_SOURCE_DIR}/../shared/version.h")
    file(STRINGS "${notice}" notice_header LIMIT_COUNT 3 ENCODING UTF-8)
    list(GET notice_header 1 notice_version)
    if(NOT notice_version STREQUAL "Version: ${cab_version}")
        message(FATAL_ERROR "License header version mismatch; run scripts/generate_licenses.py before building")
    endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    set(CAB_WINDOWS_STATIC_AUTH ON CACHE BOOL "Link authorization into the Windows EXE" FORCE)
endif()
add_subdirectory(../native ${CMAKE_CURRENT_BINARY_DIR}/backend-build)
set_target_properties(CabBackend PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/backend")
if(WIN32)
    enable_language(RC)
    add_executable(ArchetypeCabDisabler WIN32 windows/main.cpp windows/app.rc ../native/backend/session.cpp)
    target_compile_definitions(cab_core PUBLIC CAB_WINDOWS_SINGLE_EXE)
    target_link_libraries(ArchetypeCabDisabler PRIVATE cab_core)
    add_library(CabPrivilegeMode OBJECT windows/authorization.c)
    target_compile_definitions(CabPrivilegeMode PRIVATE CAB_PRIVILEGED_HELPER)
    target_sources(ArchetypeCabDisabler PRIVATE $<TARGET_OBJECTS:CabPrivilegeMode>)
    set(CAB_AUTH_TARGETS CabPrivilegeMode)
    foreach(target ArchetypeCabDisabler ${CAB_AUTH_TARGETS})
        cab_optimize(${target})
    endforeach()
    target_include_directories(ArchetypeCabDisabler PRIVATE windows)
    target_compile_definitions(ArchetypeCabDisabler PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(ArchetypeCabDisabler PRIVATE ole32 shell32 uuid comctl32 uxtheme user32 gdi32 advapi32)
    if(MSVC)
        set_property(TARGET ArchetypeCabDisabler ${CAB_AUTH_TARGETS} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(ArchetypeCabDisabler PRIVATE /W4 /utf-8)
        target_link_options(ArchetypeCabDisabler PRIVATE /MANIFEST:NO)
    else()
        target_compile_options(ArchetypeCabDisabler PRIVATE -Wall -Wextra)
        target_link_options(ArchetypeCabDisabler PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
elseif(APPLE)
    set_target_properties(CabBackend PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    add_executable(ArchetypeCabDisabler macos/main.mm)
    target_compile_options(ArchetypeCabDisabler PRIVATE -fobjc-arc -fblocks -Wall -Wextra -Wunguarded-availability -Werror=unguarded-availability)
    target_link_libraries(ArchetypeCabDisabler PRIVATE "-framework AppKit" "-framework Foundation")
    add_executable(CabPrivilege macos/authorization.c)
    target_compile_definitions(CabPrivilege PRIVATE CAB_PRIVILEGED_HELPER)
    add_library(CabAuthorization SHARED macos/authorization.c)
    set_target_properties(CabAuthorization PROPERTIES PREFIX "" SUFFIX ".dylib")
    set_target_properties(ArchetypeCabDisabler CabPrivilege PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
    set_target_properties(CabAuthorization PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
    target_link_libraries(CabAuthorization PRIVATE "-framework Security")
    foreach(target ArchetypeCabDisabler CabPrivilege CabAuthorization)
        cab_optimize(${target})
    endforeach()
else()
    message(FATAL_ERROR "Native frontend targets are Windows and macOS only")
endif()

set_target_properties(ArchetypeCabDisabler PROPERTIES OUTPUT_NAME "Archetype Cab Disabler")
