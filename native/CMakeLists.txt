# Apply the same size policy to the engine, parser and frontend targets.
function(cab_optimize target)
  if(MSVC)
    target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:/Os>" /Gy /Gw)
    if(CAB_CLANG_SIZE_LTO)
      target_compile_options(${target} PRIVATE /clang:-Oz /clang:-flto=thin)
      target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
    endif()
    if(CAB_WINDOWS_SYSTEM_UCRT)
      target_link_options(${target} PRIVATE /NODEFAULTLIB:libucrt.lib ucrt.lib)
    endif()
  else()
    if(APPLE)
      target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-Oz>" "$<$<NOT:$<CONFIG:Debug>>:-flto=thin>")
      target_link_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-flto=thin>" "$<$<NOT:$<CONFIG:Debug>>:-Wl,-x>")
    elseif(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      if(CAB_GCC_LTO)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
        target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:-flto-partition=one>")
      endif()
      target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-Oz>")
      target_link_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-s>")
    else()
      target_compile_options(${target} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-Os>")
    endif()
    if(NOT APPLE)
      target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
    endif()
    if(APPLE)
      target_link_options(${target} PRIVATE -Wl,-dead_strip)
    elseif(MINGW)
      target_link_options(${target} PRIVATE -Wl,--gc-sections)
    endif()
  endif()
endfunction()
foreach(target cab_expat cab_core CabBackend)
  cab_optimize(${target})
endforeach()

  enable_language(RC)
  target_sources(CabBackend PRIVATE backend/windows.rc)
  target_include_directories(CabBackend PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../frontends/windows)
  target_compile_definitions(cab_core PUBLIC _WIN32_WINNT=0x0A00)
  target_link_libraries(cab_core PUBLIC bcrypt shell32 ole32 uuid)
  if(CAB_WINDOWS_STATIC_AUTH)
    add_library(cab_windows_auth STATIC ../frontends/windows/authorization.c)
    target_compile_definitions(cab_windows_auth PRIVATE CAB_WINDOWS_STATIC_AUTH)
    target_compile_definitions(cab_core PUBLIC CAB_WINDOWS_STATIC_AUTH)
    target_link_libraries(cab_windows_auth PUBLIC shell32 advapi32 ole32)
    target_link_libraries(cab_core PUBLIC cab_windows_auth)
    cab_optimize(cab_windows_auth)
    if(MSVC)
      set_property(TARGET cab_windows_auth PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  endif()
  if(MINGW)
    target_link_options(CabBackend PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    set_property(TARGET CabBackend cab_core cab_expat PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(CabBackend PRIVATE /MANIFEST:NO)
  endif()
elseif(NOT APPLE)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  target_link_libraries(cab_core PUBLIC OpenSSL::Crypto ${CMAKE_DL_LIBS} Threads::Threads)
endif()

    target_link_libraries(CabTest_${name} PRIVATE cab_core)
    cab_optimize(CabTest_${name})
    if(MINGW)

      -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/test-data/${name}
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.cmake)
  endforeach()
endif()
